\newtheorem{theorem}{Theorem}

\begin{theorem}[CLR]
For all $k,\Delta,R,H$ there exists $M$ such that any $\Delta$-bounded graph with $H$-cycle complexity $>M$ has at least two distinct radius-$R$ FO$^k$ types.
\end{theorem}

\begin{theorem}[IECP]
There exist $c,\alpha,\varepsilon>0$ such that if $\int_0^T |\nabla u|\infty dt \ge \alpha T$, then $\log N\varepsilon(\mathcal R(T)) \ge cT$.
\end{theorem}

\begin{theorem}[Final Wall]
For all $k,\Delta$ there exists $C$ such that FO$^k$-homogeneous graphs have bounded overlap complexity $\le C$.
\end{theorem}
